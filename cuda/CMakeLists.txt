cmake_minimum_required(VERSION 3.24)
project(uq_cuda LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
add_library(uq_decode STATIC uq_decode.cu)
target_include_directories(uq_decode PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(uq_decode PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3 -lineinfo>)
set_target_properties(uq_decode PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(test_vectors test_vectors.cu)
target_link_libraries(test_vectors uq_decode)
enable_testing()
add_test(NAME vectors COMMAND test_vectors ${CMAKE_CURRENT_SOURCE_DIR}/vectors 1e-3)
