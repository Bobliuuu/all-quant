cmake_minimum_required(VERSION 3.20)
project(uq_sycl CXX)
# Configure with the DPC++ compiler, e.g.  cmake -S . -B build -DCMAKE_CXX_COMPILER=clang++ -G Ninja
# Non-Intel targets: -DUQ_NO_XMX=ON (joint_matrix path is Intel-XMX only).
option(UQ_NO_XMX "disable the Intel XMX joint_matrix path" OFF)
set(CMAKE_CXX_STANDARD 17)
foreach(t test_vectors bench probe)
  add_executable(${t} ${t}.cpp)
  target_compile_options(${t} PRIVATE -fsycl -O3)
  target_link_options(${t} PRIVATE -fsycl)
  if(UQ_NO_XMX)
    target_compile_definitions(${t} PRIVATE UQ_NO_XMX)
  endif()
endforeach()
enable_testing()
add_test(NAME vectors_portable COMMAND test_vectors ${CMAKE_CURRENT_SOURCE_DIR}/vectors --sweep)
if(NOT UQ_NO_XMX)
  add_test(NAME vectors_xmx COMMAND test_vectors ${CMAKE_CURRENT_SOURCE_DIR}/vectors --path xmx --sweep)
endif()
